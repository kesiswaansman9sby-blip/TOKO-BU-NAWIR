        function handleBarcodeScan(event) {
            // Scanner barcode selalu mengirimkan input diakhiri tombol 'Enter'
            if (event.key === 'Enter') {
                const term = event.target.value.trim().toLowerCase();
                if (term === '') return;
                
                // Cari apakah ada ID Produk yang cocok persis (Scan Barcode)
                let prod = db.produk.find(p => p.ID_Produk.toLowerCase() === term);
                
                if (prod) {
                    addToCart(prod.ID_Produk);
                    event.target.value = ''; // Kosongkan input setelah berhasil scan
                    filterPOS(); // Reset list produk
                } else {
                    // Jika enter ditekan tapi bukan scan (pencarian manual enter)
                    filterPOS();
                }
            } else {
                // Pencarian manual mengetik huruf
                filterPOS();
            }
        }

        function addToCart(id) {
            const prod = db.produk.find(p => p.ID_Produk === id);
            if(!prod) return;
            if(parseInt(prod.Stok) <= 0) { Swal.fire('Gagal', 'Stok produk ini habis!', 'warning'); return; }

            const existing = cart.find(c => c.ID_Produk === id);
            if(existing) {
                if(existing.qty >= prod.Stok) return Swal.fire('Gagal', 'Kuantitas melebihi stok tersedia!', 'warning');
                existing.qty++;
            } else {
                cart.push({ ...prod, qty: 1 });
            }
            renderCart();
        }

        function updateQty(id, delta) {
            const item = cart.find(c => c.ID_Produk === id);
            if(!item) return;
            const prod = db.produk.find(p => p.ID_Produk === id);
            
            item.qty += delta;
            if(item.qty <= 0) {
                cart = cart.filter(c => c.ID_Produk !== id);
            } else if(item.qty > prod.Stok) {
                item.qty = prod.Stok;
            }
            
            renderCart();
        }

        function renderCart() {
            const list = document.getElementById('cartItemsList');
            list.innerHTML = '';
            let subtotal = 0;

            if(cart.length === 0) {
                list.innerHTML = `
                <div class="text-center text-muted mt-5 py-4">
                    <i class="fa-solid fa-box-open fa-3x mb-3" style="color: #cbd5e1;"></i>
                    <p>Keranjang masih kosong</p>
                </div>`;
            } else {
                cart.forEach(c => {
                    const st = c.Harga_Jual * c.qty;
                    subtotal += st;
                    list.innerHTML += `
                        <div class="cart-item">
                            <div style="flex:2">
                                <div class="fw-bold text-truncate text-dark" style="max-width: 130px;" title="${c.Nama_Produk}">${c.Nama_Produk}</div>
                                <div class="small text-muted">${formatRupiah(c.Harga_Jual)}</div>
                            </div>
                            <div class="cart-item-qty" style="flex:1; justify-content:center;">
                                <button onclick="updateQty('${c.ID_Produk}', -1)"><i class="fa-solid fa-minus" style="font-size:10px;"></i></button>
                                <span class="fw-bold px-1">${c.qty}</span>
                                <button onclick="updateQty('${c.ID_Produk}', 1)"><i class="fa-solid fa-plus" style="font-size:10px;"></i></button>
                            </div>
                            <div class="text-end fw-bold text-primary" style="flex:1">${formatRupiah(st)}</div>
                        </div>
                    `;
                });
            }
            
            document.getElementById('cartSubtotal').dataset.raw = subtotal;
            document.getElementById('cartSubtotal').innerText = formatRupiah(subtotal);
            calculateCart();
        }

        function calculateCart() {
            const subtotal = parseInt(document.getElementById('cartSubtotal').dataset.raw) || 0;
            let diskon = parseInt(document.getElementById('cartDiskon').value);
            if(isNaN(diskon) || diskon < 0) diskon = 0;
            
            const total = subtotal - diskon;
            let tunai = parseInt(document.getElementById('cartTunai').value);
            if(isNaN(tunai)) tunai = 0;
            
            const kembalian = tunai - total;

            document.getElementById('cartTotal').innerText = formatRupiah(total);
            
            const elKembalian = document.getElementById('cartKembalian');
            elKembalian.innerText = formatRupiah(kembalian);
            
            if (kembalian < 0) {
                elKembalian.classList.remove('text-primary');
                elKembalian.classList.add('text-danger');
            } else {
                elKembalian.classList.remove('text-danger');
                elKembalian.classList.add('text-primary');
            }
            
            const btnBayar = document.getElementById('btnBayar');
            btnBayar.disabled = (cart.length === 0 || kembalian < 0 || total < 0);
        }

        function processCheckout() {
            const subtotal = parseInt(document.getElementById('cartSubtotal').dataset.raw) || 0;
            const diskon = parseInt(document.getElementById('cartDiskon').value) || 0;
            const totalBayar = subtotal - diskon;
            const tunai = parseInt(document.getElementById('cartTunai').value) || 0;
            const kembalian = tunai - totalBayar;

            const trxID = generateID('TRX');
            let totalProfit = 0;
            
            const details = cart.map(c => {
                const sub = c.Harga_Jual * c.qty;
                const modal = c.Harga_Beli * c.qty;
                const profit = sub - modal;
                totalProfit += profit;
                return { ID_Produk: c.ID_Produk, Nama_Produk: c.Nama_Produk, Jumlah: c.qty, Subtotal: sub, Profit: profit };
            });

            const dataTrx = { 
                ID_Transaksi: trxID, Total_Bayar: totalBayar, Diskon: diskon, 
                Uang_Tunai: tunai, Kembalian: kembalian, 
                Nama_Kasir: currentUser.nama, Total_Profit: totalProfit - diskon 
            };

            startLoading('Memproses Pembayaran...');
            const btn = document.getElementById('btnBayar');
            btn.disabled = true;

            const handleSuccessCheckout = () => {
                Swal.fire({
                    title: 'Pembayaran Berhasil!',
                    html: `Kembalian: <br><b class="fs-1 text-primary">${formatRupiah(kembalian)}</b>`,
                    icon: 'success',
                    showCancelButton: true,
                    confirmButtonColor: '#16a34a',
                    cancelButtonColor: '#6b7280',
                    confirmButtonText: '<i class="fa-solid fa-print"></i> Cetak Struk',
                    cancelButtonText: 'Tutup Saja'
                }).then((result) => {
                    if (result.isConfirmed) {
                        cetakStruk(dataTrx, details);
                    }
                    resetCart(); 
                    loadDataServer();
                });
            };

            if(typeof google === 'undefined') {
                // SIMULASI
                setTimeout(() => {
                    endLoading();
                    mockDB.transaksi.push({ ...dataTrx, Waktu: new Date().toISOString() });
                    details.forEach(d => {
                        let prod = mockDB.produk.find(p => p.ID_Produk === d.ID_Produk);
                        if(prod) prod.Stok -= d.Jumlah;
                    });
                    handleSuccessCheckout();
                }, 800);
            } else {
                // REAL SERVER
                google.script.run
                    .withSuccessHandler(res => {
                        endLoading();
                        if(res.success) {
                            handleSuccessCheckout();
                        } else { 
                            Swal.fire('Gagal', res.message, 'error'); 
                            btn.disabled = false; 
                        }
                    })
                    .withFailureHandler(err => {
                        endLoading(); 
                        btn.disabled = false;
                        Swal.fire('Error', err.message, 'error');
                    })
                    .simpanTransaksi(dataTrx, details);
            }
        }

        function renderTableProduk() {
            const tbody = document.getElementById('tbodyProduk');
            const term = document.getElementById('searchProduk').value.toLowerCase();
            tbody.innerHTML = '';
            
            let filtered = db.produk;
            if(term) filtered = filtered.filter(p => p.Nama_Produk.toLowerCase().includes(term) || p.ID_Produk.toLowerCase().includes(term));

            if(filtered.length===0) return tbody.innerHTML = '<tr><td colspan="7" class="text-center py-4 text-muted">Tidak ada data produk</td></tr>';

            filtered.forEach(p => {
                const img = p.URL_Gambar || 'https://via.placeholder.com/50';
                tbody.innerHTML += `
                    <tr>
                        <td><img src="${img}" width="45" height="45" style="object-fit:cover; border-radius:8px; border:1px solid #e2e8f0;"></td>
                        <td class="text-muted small">${p.ID_Produk}</td>
                        <td class="fw-bold text-start text-dark">${p.Nama_Produk}</td>
                        <td><span class="badge bg-light text-dark border">${p.Kategori}</span></td>
                        <td class="fw-bold">${p.Stok}</td>
                        <td class="text-primary fw-bold">${formatRupiah(p.Harga_Jual)}</td>
                        <td>
                            <button class="btn btn-sm btn-outline-danger px-2 py-1 shadow-sm" onclick="deleteProduk('${p.ID_Produk}')"><i class="fa-solid fa-trash"></i></button>
                        </td>
                    </tr>
                `;
            });
        }

        const modalProduk = new bootstrap.Modal(document.getElementById('modalProduk'));
        function showModalProduk() {

            document.getElementById('p_id').value = ''; // Biarkan kosong agar bisa di scan barcode

        function simpanProduk(e) {
            e.preventDefault();
            const data = {
                ID_Produk: document.getElementById('p_id').value || generateID('BR'),
                Nama_Produk: document.getElementById('p_nama').value,
                Kategori: document.getElementById('p_kategori').value,
                URL_Gambar: document.getElementById('p_gambar').value,
                Harga_Beli: document.getElementById('p_hbeli').value,
                Harga_Jual: document.getElementById('p_hjual').value,
                Stok: document.getElementById('p_stok').value
            };
            
            const btn = document.getElementById('btnSimpanProduk');
            btn.disabled = true; 
            startLoading('Menyimpan Data...');

            if(typeof google === 'undefined') {
                setTimeout(() => {
                    endLoading(); modalProduk.hide(); btn.disabled=false;
                    mockDB.produk.push(data);
                    Swal.fire('Sukses', 'Data Produk ditambahkan ke memori', 'success').then(() => loadDataServer());
                }, 600);
            } else {
                google.script.run
                    .withSuccessHandler(res => {
                        endLoading(); modalProduk.hide(); btn.disabled=false;
                        if(res.success) { Swal.fire('Sukses', res.message, 'success'); loadDataServer(); }
                        else Swal.fire('Gagal', res.message, 'error');
                    })
                    .withFailureHandler(err => {
                        endLoading(); btn.disabled=false;
                        Swal.fire('Error', err.message, 'error');
                    }).simpanProdukBaru(data);
            }
        }

        function deleteProduk(id) {
            Swal.fire({ title: 'Hapus Produk?', text: 'Data akan dihapus permanen', icon: 'warning', showCancelButton: true, confirmButtonText: 'Ya, Hapus', confirmButtonColor: '#dc3545' })
            .then((res) => {
                if(res.isConfirmed) {
                    startLoading('Menghapus Data...');
                    if(typeof google === 'undefined') {
                        setTimeout(()=>{
                            endLoading();
                            mockDB.produk = mockDB.produk.filter(p => p.ID_Produk !== id);
                            Swal.fire('Dihapus', 'Data dihapus dari memori', 'success').then(()=>loadDataServer());
                        },500);
                    } else {
                        google.script.run
                        .withSuccessHandler(r => {
                            endLoading();
                            if(r.success) { Swal.fire('Sukses', r.message, 'success'); loadDataServer(); }
                            else Swal.fire('Gagal', r.message, 'error');
                        })
                        .withFailureHandler(e => { endLoading(); Swal.fire('Error', e.message, 'error'); })
                        .hapusProduk(id);
                    }
                }
            });
        }

        function renderTableTransaksi() {
            const tbody = document.getElementById('tbodyTransaksi');
            tbody.innerHTML = '';
            if(!db.transaksi || db.transaksi.length === 0) {
                tbody.innerHTML = '<tr><td colspan="4" class="text-center py-4 text-muted">Belum ada riwayat transaksi</td></tr>'; return;
            }
            
            const sorted = [...db.transaksi].reverse();
            sorted.forEach(t => {
                const dateRaw = t.Waktu ? t.Waktu.split('T') : ['-','-'];
                const waktuFmt = `${dateRaw[0]} ${dateRaw[1] ? dateRaw[1].substring(0,5) : ''}`;
                tbody.innerHTML += `
                    <tr>
                        <td class="text-muted small fw-bold">${t.ID_Transaksi}</td>
                        <td>${waktuFmt}</td>
                        <td class="text-primary fw-bold">${formatRupiah(t.Total_Bayar)}</td>
                        <td><span class="badge bg-light text-dark border px-2 py-1"><i class="fa-solid fa-circle-user text-muted"></i> ${t.Nama_Kasir}</span></td>
                    </tr>
                `;
            });
        }
